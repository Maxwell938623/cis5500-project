Query 1: Top 10 Busiest Stations by Ridership between 2020-2024
Description: This query aggregates total ridership across all five years and payment methods for each station and returns the ten stations with the highest amount of total foot traffic. This data will be used for the Station Safety Index page's ranking feature and gives context behind security enforcement data compared to ridership.
SELECT station_complex, borough, SUM(ridership) AS total_ridership
FROM ridership
GROUP BY station_complex, borough
ORDER BY total_ridership DESC
LIMIT 10;
	Query 2: Quarterly System-Wide Fare Evasion Rate Over Time
Description: This query retrieves the full time series of the history of fare evasion percentages and their corresponding margins of error in chronological order. This data will be presented on the Financial Impact Analysis page's historical trend chart.
SELECT year, quarter, ROUND(fare_evasion * 100, 2) AS evasion_pct, ROUND(margin_of_error * 100, 2) AS margin_of_error_pct
FROM fairevasion
WHERE fare_evasion IS NOT NULL
ORDER BY year ASC, quarter ASC;
	Query 3: Borough-Level Total Ridership by Year
Description: This query summarizes total ridership per borough per year and is presented on the Borough-Level Equity Comparison page using side-by-side bar charts. We filter out the NULL boroughs to make sure that only the labeled records are included.
SELECT borough, year, SUM(ridership) AS total_ridership
FROM ridership
WHERE borough IS NOT NULL
GROUP BY borough, year
ORDER BY borough, year;
	Query 4: ADA-Accessible Stations per Borough
Description: This query counts the fully ADA-accessible stations (ADA = 1) grouped by borough using the stationcoords table. This can provide additional information for the Borough-Level Equity page with accessibility context.
SELECT borough, COUNT(*) AS accessible_stations
FROM stationcoords
WHERE ADA = 1
GROUP BY borough
ORDER BY accessible_stations DESC;
	Query 5: Payment Method Share by Borough
Description: This query breaks down total ridership by borough and payment method, allowing us to analyze which payment methods are the most popular by borough. This will be included on the Financial Impact Analysis page's for fare collection modalities.
SELECT borough, payment_method, SUM(ridership) AS total_ridership,
    ROUND(100.0 * SUM(ridership) / SUM(SUM(ridership)) OVER (PARTITION BY borough), 2) AS pct_of_borough
FROM ridership
WHERE borough IS NOT NULL
GROUP BY borough, payment_method
ORDER BY borough, total_ridership DESC;
	Query 6: Estimated Revenue Lost to Fare Evasion per Quarter (Complex)
Description: This query joins quarterly ridership totals with fare evasion rates to estimate the number of rides where fare evasion took place and the resulting revenue loss, assuming a $2.90 fare per ride. This is our core calculation and formula for the numbers for the Financial Impact Analysis dashboard.
SELECT r.year,  r.quarter, SUM(r.ridership) AS total_paid_rides, fe.fare_evasion AS evasion_rate,
    ROUND(SUM(r.ridership) / NULLIF(1 - fe.fare_evasion, 0)) AS est_total_boardings,
    ROUND(SUM(r.ridership) / NULLIF(1 - fe.fare_evasion, 0) * fe.fare_evasion) AS est_evaded_rides,
    ROUND(SUM(r.ridership) / NULLIF(1 - fe.fare_evasion, 0) * fe.fare_evasion * 2.90, 2) AS est_revenue_lost_usd
FROM ridership r
JOIN fairevasion fe ON r.year = fe.year AND r.quarter = fe.quarter
WHERE fe.fare_evasion IS NOT NULL
GROUP BY r.year, r.quarter, fe.fare_evasion
ORDER BY r.year, r.quarter;
	Query 7: Borough Enforcement-to-Rider Ratio (Complex)
Description: This query joins ridership data with station coordinates to compute total paid rides per borough. It is then cross-referenced with a subquery of estimated evasion volume, giving us an enforcement-intensity metric normalized per 100,000 riders. This is the main metric for the Borough-Level Equity Comparison page and addresses our goal of showing how disparate policing is relative to rider traffic.
WITH borough_ridership AS (
    SELECT r.borough, r.year, SUM(r.ridership) AS paid_rides
    FROM ridership r
    WHERE r.borough IS NOT NULL
    GROUP BY r.borough, r.year
),
borough_evasion_est AS (
    SELECT br.borough, br.year, br.paid_rides, fe_avg.avg_evasion_rate,
        ROUND(br.paid_rides / NULLIF(1 - fe_avg.avg_evasion_rate, 0) * fe_avg.avg_evasion_rate) AS est_evaded_rides
    FROM borough_ridership br
    JOIN (
        SELECT year, AVG(fare_evasion) AS avg_evasion_rate
        FROM fairevasion
        WHERE fare_evasion IS NOT NULL
        GROUP BY year
    ) fe_avg ON br.year = fe_avg.year
)
SELECT borough, year, paid_rides, est_evaded_rides,
    ROUND(100000.0 * est_evaded_rides / NULLIF(paid_rides, 0), 2) AS est_evaded_per_100k_riders
FROM borough_evasion_est
ORDER BY year DESC, est_evaded_per_100k_riders DESC;
	Query 8: Station Safety Index, Fare Class Risk Profile per Station (Complex)
Description: This query computes the percentage of rides paid with reduced-fare or other methods that correlate with higher evasion risk versus full-fare payment categories. It then ranks stations within each borough by this risk score. This provides information for the Station Safety Index page's dropdown-based station lookup and filterable risk ranking.
WITH station_fare_breakdown AS (
    SELECT station_complex_id, station_complex, borough, SUM(ridership) AS total_ridership,
        SUM(CASE WHEN fare_class_category ILIKE '%unlimited%'
              OR fare_class_category ILIKE '%full fare%'
            THEN ridership ELSE 0 END) AS full_fare_rides,
        SUM(CASE WHEN fare_class_category ILIKE '%reduced%'
              OR fare_class_category ILIKE '%student%'
              OR fare_class_category ILIKE '%senior%'
            THEN ridership ELSE 0 END) AS reduced_fare_rides
    FROM ridership
    GROUP BY station_complex_id, station_complex, borough
),
station_risk AS (
    SELECT *, ROUND(100.0 * reduced_fare_rides /  NULLIF(total_ridership, 0), 2) AS reduced_fare_pct,
        RANK() OVER (
            PARTITION BY borough
            ORDER BY 100.0 * reduced_fare_rides / NULLIF(total_ridership, 0) DESC
        ) AS borough_rank
    FROM station_fare_breakdown
)
SELECT station_complex, borough, total_ridership, reduced_fare_pct, borough_rank
FROM station_risk
ORDER BY borough, borough_rank
LIMIT 50;
	Query 9: Annual Trends of Ridership Recovery vs. Evasion Rate (Complex)
Description: This query computes the annual total ridership data with each year’s average fare evasion rate. It then calculates the annual change in percentages in both of these metrics, which provides data for our Historical Time-Series Trend Analysis feature. For example, a large difference between a recovering ridership and a high evasion rate would be worth noting.
WITH annual_stats AS (
    SELECT r.year, SUM(r.ridership) AS total_ridership, AVG(fe.fare_evasion) AS avg_evasion_rate
    FROM ridership r LEFT JOIN fairevasion fe ON r.year = fe.year AND fe.fare_evasion IS NOT NULL
    GROUP BY r.year),
with_lag AS (
    SELECT year, total_ridership, ROUND(avg_evasion_rate * 100, 3) AS evasion_pct,
        LAG(total_ridership)  OVER (ORDER BY year) AS prev_ridership,
        LAG(avg_evasion_rate) OVER (ORDER BY year) AS prev_evasion
    FROM annual_stats)
SELECT year, total_ridership, evasion_pct,
    ROUND(100.0 * (total_ridership - prev_ridership) / NULLIF(prev_ridership, 0), 2) AS yoy_ridership_change_pct,
    ROUND(100.0 * (evasion_pct/100 - prev_evasion) / NULLIF(prev_evasion, 0), 2) AS yoy_evasion_change_pct
FROM with_lag
ORDER BY year;
	Query 10: Geospatial Cluster: Highest-Traffic Stations Outside CBD by Borough (Complex)
Description: This query joins ridership data with stationcoords to pinpoint the top five non-CBD stations per borough by total ridership. It then renders this into the map using the coordinates and subway lines for the Geospatial Heatmap’s Ridership Volume layer. This gives critical information as to whether law enforcement is concentrated too much in commercial areas, and should be shifted out to outer-borough neighborhoods.
WITH station_volume AS (
    SELECT r.station_complex_id, r.station_complex, r.borough, SUM(r.ridership) AS total_ridership, AVG(r.latitude) AS latitude, AVG(r.longitude) AS longitude
    FROM ridership r
    GROUP BY r.station_complex_id, r.station_complex, r.borough),
with_coords AS (
    SELECT sv.station_complex, sv.borough, sv.total_ridership, sv.latitude, sv.longitude, sc.daytime_routes AS lines, sc.CBD AS is_cbd,
        RANK() OVER (
            PARTITION BY sv.borough
            ORDER BY sv.total_ridership DESC
        ) AS borough_rank
    FROM station_volume sv JOIN stationcoords sc ON sv.station_complex_id::text = sc.complex_id::text
    WHERE sc.CBD = FALSE
)
SELECT station_complex, borough, lines, total_ridership, latitude, longitude, borough_rank
FROM with_coords
WHERE borough_rank <= 5
ORDER BY borough, borough_rank;
